# 1.应用场景与目标

应用场景： 高校学生或求职者在正式面试前，需要针对特定岗位进行模拟实战练习，以克服紧张情绪、梳理答题逻辑。

核心解决问题：传统的“背诵面经”缺乏互动性和针对性；找真人模拟面试成本高、难以随时进行。

**大模型的角色：**

提问者： 根据用户提供的岗位和简历信息，动态生成结构化面试问题或专业技术问题。

倾听与追问者： 评估用户的回答，如果不完整或有漏洞，进行针对性的深度追问（这是体现AI智能的核心亮点）。

评估专家： 面试结束后，根据整体表现生成包含多维度的“面试诊断报告”。

```python
import os
import re
from typing import Any, Dict, List, Optional, Tuple

import streamlit as st
from dotenv import load_dotenv
from zhipuai import ZhipuAI


SYSTEM_PROMPT_PATH = "prompts/system_prompt.md"
MODEL_NAME = "glm-4.7"
ASR_MODEL_NAME = "glm-asr"
MAX_RESUME_CHARS = 6000
MIN_AUDIO_STREAMLIT_VERSION = (1, 55, 0)
SUPPORTED_AUDIO_SUFFIXES = (".wav", ".mp3", ".m4a", ".aac", ".ogg", ".webm")

DIFFICULTY_RULES = {
    "简单": "问题以基础概念和常见场景为主，允许给少量提示，避免过度刁钻。",
    "普通": "覆盖基础与进阶，既考察原理也考察工程实践，适度追问。",
    "地狱": "重点考察边界条件、性能权衡、故障恢复与架构深度，追问更严格。",
}

EVALUATION_SYSTEM_PROMPT = """
你是一位严谨的技术面试评估专家。请基于完整对话输出结构化 Markdown 报告。
报告必须包含以下部分并使用清晰标题：
1) 综合评分（0-100）+ 一句话结论
2) 能力雷达维度评分（逻辑清晰度、专业知识准确度、沟通表达能力，分别给出 0-100）
3) 逐题点评（每题给出亮点、不足、改进建议）
4) 参考满分回答（逐题给出可借鉴答案，简洁但专业）
要求：客观、具体、可执行，避免空泛夸奖。
""".strip()


def load_prompt(file_path: str, **kwargs: Any) -> str:
    if not os.path.exists(file_path):
        st.error(f"找不到 Prompt 文件：{file_path}")
        return ""

    with open(file_path, "r", encoding="utf-8") as f:
        content = f.read()

    for key, value in kwargs.items():
        content = content.replace(f"{{{{{key}}}}}", str(value))
    return content


def parse_version_tuple(version_text: str) -> Tuple[int, int, int]:
    parts = version_text.split(".")
    parsed = []
    for idx in range(3):
        if idx >= len(parts):
            parsed.append(0)
            continue
        digits = []
        for ch in parts[idx]:
            if ch.isdigit():
                digits.append(ch)
            else:
                break
        parsed.append(int("".join(digits) or "0"))
    return parsed[0], parsed[1], parsed[2]


def is_audio_chat_supported() -> bool:
    return parse_version_tuple(st.__version__) >= MIN_AUDIO_STREAMLIT_VERSION


def init_state() -> None:
    defaults = {
        "messages": [],
        "interview_started": False,
        "asked_count": 0,
        "target_count": 4,
        "difficulty": "普通",
        "interview_ended": False,
        "report_markdown": "",
        "resume_source": "manual",
        "resume_filename": "",
        "resume_text": "",
        "voice_enabled": True,
        "asr_available": True,
    }
    for key, value in defaults.items():
        if key not in st.session_state:
            st.session_state[key] = value


def create_client(api_key: Optional[str]) -> Optional[ZhipuAI]:
    if not api_key:
        return None
    try:
        return ZhipuAI(api_key=api_key)
    except Exception as e:  # noqa: BLE001
        st.error(f"初始化模型客户端失败：{e}")
        return None


def extract_delta_text(chunk: Any) -> str:
    if not getattr(chunk, "choices", None):
        return ""
    delta = chunk.choices[0].delta
    content = getattr(delta, "content", "")
    if isinstance(content, list):
        return "".join(str(item) for item in content if item)
    return str(content or "")


def extract_message_text(response: Any) -> str:
    if not getattr(response, "choices", None):
        return ""
    message = response.choices[0].message
    content = getattr(message, "content", "")
    if isinstance(content, list):
        return "".join(str(item) for item in content if item)
    return str(content or "").strip()


def sanitize_text(text: str) -> str:
    clean_text = re.sub(r"[ \t]+", " ", text or "")
    clean_text = re.sub(r"\n{3,}", "\n\n", clean_text)
    return clean_text.strip()


def extract_pdf_text(uploaded_file: Any) -> str:
    try:
        from pypdf import PdfReader
    except ImportError:
        st.error("缺少 `pypdf` 依赖，请先安装：`pip install pypdf`。")
        return ""

    try:
        uploaded_file.seek(0)
        reader = PdfReader(uploaded_file)
        page_texts = []
        for page in reader.pages:
            page_texts.append(page.extract_text() or "")
        return sanitize_text("\n".join(page_texts))
    except Exception as e:  # noqa: BLE001
        st.error(f"解析 PDF 失败：{e}")
        return ""


def resolve_resume_info(uploaded_pdf: Any, manual_summary: str) -> Tuple[str, str, str]:
    source = "manual"
    filename = ""
    resume_text = sanitize_text(manual_summary)

    if uploaded_pdf is not None:
        filename = getattr(uploaded_pdf, "name", "")
        pdf_text = extract_pdf_text(uploaded_pdf)
        if pdf_text:
            source = "pdf"
            resume_text = pdf_text
        elif resume_text:
            st.warning("PDF 解析失败，已自动回退到手动摘要内容。")
        else:
            st.warning("未能从 PDF 提取到文本内容，请补充手动摘要。")

    if not resume_text:
        resume_text = "未提供简历信息。"

    if len(resume_text) > MAX_RESUME_CHARS:
        resume_text = resume_text[:MAX_RESUME_CHARS]
        st.info(f"简历内容超过 {MAX_RESUME_CHARS} 字符，已自动截断。")

    return resume_text, source, filename


def get_payload_value(payload: Any, key: str) -> Any:
    if payload is None:
        return None
    if isinstance(payload, dict):
        return payload.get(key)
    return getattr(payload, key, None)


def normalize_chat_input(payload: Any) -> Tuple[str, Optional[Any]]:
    if payload is None:
        return "", None
    if isinstance(payload, str):
        return payload.strip(), None

    text = ""
    for key in ("text", "message", "content", "value"):
        value = get_payload_value(payload, key)
        if isinstance(value, str) and value.strip():
            text = value.strip()
            break

    audio_file = get_payload_value(payload, "audio")
    if audio_file is not None:
        return text, audio_file

    files = get_payload_value(payload, "files")
    if files is None:
        single_file = get_payload_value(payload, "file")
        files = [single_file] if single_file is not None else []
    elif not isinstance(files, list):
        files = [files]

    for file_obj in files:
        if file_obj is None:
            continue
        file_name = str(getattr(file_obj, "name", "")).lower()
        file_type = str(getattr(file_obj, "type", ""))
        if file_type.startswith("audio/") or file_name.endswith(SUPPORTED_AUDIO_SUFFIXES):
            audio_file = file_obj
            break

    return text, audio_file


def find_text_in_obj(obj: Any) -> str:
    if obj is None:
        return ""
    if isinstance(obj, str):
        return obj.strip()
    if isinstance(obj, dict):
        preferred_keys = ("text", "transcript", "content", "result")
        for key in preferred_keys:
            if key in obj:
                value = find_text_in_obj(obj[key])
                if value:
                    return value
        for value in obj.values():
            text = find_text_in_obj(value)
            if text:
                return text
        return ""
    if isinstance(obj, list):
        for item in obj:
            text = find_text_in_obj(item)
            if text:
                return text
        return ""

    for key in ("text", "transcript", "content", "result"):
        value = getattr(obj, key, None)
        text = find_text_in_obj(value)
        if text:
            return text

    model_dump_fn = getattr(obj, "model_dump", None)
    if callable(model_dump_fn):
        try:
            return find_text_in_obj(model_dump_fn())
        except Exception:  # noqa: BLE001
            return ""
    return ""


def is_asr_unavailable_error(error_text: str) -> bool:
    normalized = (error_text or "").lower()
    markers = [
        '"code":"1113"',
        "code:1113",
        "1113",
        "余额不足",
        "无可用资源包",
    ]
    return any(marker in normalized for marker in markers)


def transcribe_audio_to_text(client: ZhipuAI, audio_file: Any) -> str:
    last_error = None
    file_name = getattr(audio_file, "name", "audio_input.webm")
    file_type = getattr(audio_file, "type", "application/octet-stream")

    audio_bytes = b""
    try:
        if hasattr(audio_file, "getvalue"):
            audio_bytes = audio_file.getvalue()
        else:
            audio_file.seek(0)
            audio_bytes = audio_file.read()
    except Exception:  # noqa: BLE001
        audio_bytes = b""

    candidates = [{"file": audio_file}]
    if audio_bytes:
        candidates.append({"file": (file_name, audio_bytes, file_type)})
        candidates.append({"file": (file_name, audio_bytes)})

    for candidate in candidates:
        try:
            response = client.audio.transcriptions.create(
                model=ASR_MODEL_NAME,
                **candidate,
            )
            text = find_text_in_obj(response)
            if text:
                return text
        except Exception as e:  # noqa: BLE001
            error_text = str(e)
            if is_asr_unavailable_error(error_text):
                st.session_state.asr_available = False
                st.session_state.voice_enabled = False
                st.warning("语音功能暂不可用")
                return ""
            last_error = e

    if last_error:
        st.error(f"语音识别失败：{last_error}")
    else:
        st.error("语音识别失败：未返回有效转写文本。")
    return ""


def stream_interviewer_reply(
    client: ZhipuAI,
    messages: List[Dict[str, str]],
    placeholder: Any,
) -> str:
    full_response = ""
    stream = client.chat.completions.create(
        model=MODEL_NAME,
        messages=messages,
        stream=True,
    )
    for chunk in stream:
        delta_content = extract_delta_text(chunk)
        if delta_content:
            full_response += delta_content
            placeholder.markdown(full_response + "▌")
    placeholder.markdown(full_response)
    return full_response.strip()


def generate_report(
    client: ZhipuAI,
    role: str,
    difficulty: str,
    target_count: int,
    messages: List[Dict[str, str]],
) -> str:
    conversation_lines = []
    for msg in messages:
        if msg["role"] == "system":
            continue
        speaker = "面试官" if msg["role"] == "assistant" else "候选人"
        conversation_lines.append(f"{speaker}：{msg['content']}")

    report_request = f"""
请基于以下面试记录给出诊断报告。
目标岗位：{role}
难度：{difficulty}
目标题数：{target_count}

面试记录：
{chr(10).join(conversation_lines)}
""".strip()

    response = client.chat.completions.create(
        model=MODEL_NAME,
        messages=[
            {"role": "system", "content": EVALUATION_SYSTEM_PROMPT},
            {"role": "user", "content": report_request},
        ],
    )
    return extract_message_text(response)


def ensure_prerequisites(client: Optional[ZhipuAI]) -> bool:
    if client is None:
        st.error("未检测到可用的 `ZHIPUAI_API_KEY`，请在 `.env` 中配置后重试。")
        return False
    if not os.path.exists(SYSTEM_PROMPT_PATH):
        st.error(f"缺少系统 Prompt 文件：`{SYSTEM_PROMPT_PATH}`")
        return False
    return True


def start_interview(
    role: str,
    resume: str,
    difficulty: str,
    question_count: int,
) -> None:
    full_system_prompt = load_prompt(
        SYSTEM_PROMPT_PATH,
        target_role=role,
        resume_info=resume,
        difficulty=difficulty,
        target_question_count=question_count,
        difficulty_rule=DIFFICULTY_RULES[difficulty],
    )
    if not full_system_prompt:
        return

    st.session_state.messages = [{"role": "system", "content": full_system_prompt}]
    st.session_state.messages.append(
        {"role": "assistant", "content": "你好，我是本次面试的面试官，准备好了吗？"}
    )
    st.session_state.interview_started = True
    st.session_state.interview_ended = False
    st.session_state.asked_count = 0
    st.session_state.target_count = question_count
    st.session_state.difficulty = difficulty
    st.session_state.report_markdown = ""
    st.session_state.resume_text = resume


def end_interview(client: ZhipuAI, role: str) -> None:
    if st.session_state.interview_ended or not st.session_state.interview_started:
        return
    try:
        with st.spinner("正在生成面试评估报告..."):
            report = generate_report(
                client=client,
                role=role,
                difficulty=st.session_state.difficulty,
                target_count=st.session_state.target_count,
                messages=st.session_state.messages,
            )
        st.session_state.report_markdown = report or "报告生成失败，请稍后重试。"
        st.session_state.interview_ended = True
    except Exception as e:  # noqa: BLE001
        st.session_state.report_markdown = ""
        st.error(f"生成报告失败：{e}")


load_dotenv()
api_key = os.getenv("ZHIPUAI_API_KEY")
client = create_client(api_key)

st.title("AI 模拟面试")
st.caption("按“准备配置 → 一问一答 → 结束复盘”进行完整练习。")

init_state()
audio_supported = is_audio_chat_supported()

with st.sidebar:
    st.header("面试设置")
    role = st.text_input("目标岗位", value="前端开发工程师")
    resume_pdf = st.file_uploader("上传简历 PDF（可选）", type=["pdf"])
    if resume_pdf is not None:
        st.caption(f"已上传：{resume_pdf.name}")
        st.session_state.resume_filename = resume_pdf.name
        try:
            st.toast(f"已上传简历：{resume_pdf.name}", icon="✅")
        except Exception:  # noqa: BLE001
            pass
    resume_manual_summary = st.text_area("简历摘要（兜底，可选）", value="熟练掌握 Vue3")

    difficulty = st.selectbox("面试难度", options=["简单", "普通", "地狱"], index=1)
    question_count = st.slider("面试题数", min_value=3, max_value=5, value=4, step=1)
    if not st.session_state.asr_available:
        st.session_state.voice_enabled = False
    st.session_state.voice_enabled = st.toggle(
        "启用语音输入",
        value=st.session_state.voice_enabled,
        disabled=not st.session_state.asr_available,
    )
    start_btn = st.button("开始 / 重置面试", type="primary")
    end_interview_button = st.button(
        "结束面试",
        disabled=not st.session_state.interview_started or st.session_state.interview_ended,
    )
    st.divider()
    st.write(f"当前进度：{st.session_state.asked_count}/{st.session_state.target_count}")
    st.write(
        f"当前状态：{'已结束' if st.session_state.interview_ended else '进行中' if st.session_state.interview_started else '未开始'}"
    )
    if st.session_state.resume_source == "pdf" and st.session_state.resume_filename:
        st.caption(f"简历来源：PDF（{st.session_state.resume_filename}）")
    elif st.session_state.resume_source == "manual":
        st.caption("简历来源：手动摘要")

if st.session_state.voice_enabled and not audio_supported:
    st.warning(
        "当前 Streamlit 版本不支持聊天框语音输入。"
        f"请升级到 >= {MIN_AUDIO_STREAMLIT_VERSION[0]}.{MIN_AUDIO_STREAMLIT_VERSION[1]}，"
        "当前将回退为文本输入。"
    )

if start_btn:
    if ensure_prerequisites(client):
        resume_info, resume_source, resume_filename = resolve_resume_info(
            uploaded_pdf=resume_pdf,
            manual_summary=resume_manual_summary,
        )
        st.session_state.resume_source = resume_source
        st.session_state.resume_filename = resume_filename
        start_interview(
            role=role,
            resume=resume_info,
            difficulty=difficulty,
            question_count=question_count,
        )

if end_interview_button and ensure_prerequisites(client):
    end_interview(client=client, role=role)

if st.session_state.interview_started:
    for msg in st.session_state.messages:
        if msg["role"] == "system":
            continue
        with st.chat_message(msg["role"]):
            st.markdown(msg["content"])

if st.session_state.interview_ended and st.session_state.report_markdown:
    st.subheader("面试诊断报告")
    st.markdown(st.session_state.report_markdown)

chat_disabled = not st.session_state.interview_started or st.session_state.interview_ended
chat_label = "请输入你的回答（支持文本/语音）..." if st.session_state.voice_enabled else "请输入你的回答..."
chat_kwargs = {"disabled": chat_disabled}
if st.session_state.voice_enabled and audio_supported:
    chat_kwargs["accept_audio"] = True
chat_payload = st.chat_input(chat_label, **chat_kwargs)

if chat_payload is not None and ensure_prerequisites(client):
    user_input, audio_file = normalize_chat_input(chat_payload)

    if not user_input and audio_file is not None:
        with st.spinner("正在进行语音识别..."):
            user_input = transcribe_audio_to_text(client=client, audio_file=audio_file)

    if not user_input:
        if audio_file is None or st.session_state.asr_available:
            st.warning("未检测到有效输入，请输入文本或重新录音后提交。")
    else:
        with st.chat_message("user"):
            st.markdown(user_input)
        st.session_state.messages.append({"role": "user", "content": user_input})

        if st.session_state.asked_count >= st.session_state.target_count:
            end_interview(client=client, role=role)
        else:
            with st.chat_message("assistant"):
                message_placeholder = st.empty()
                try:
                    reply = stream_interviewer_reply(
                        client=client,
                        messages=st.session_state.messages,
                        placeholder=message_placeholder,
                    )
                    if not reply:
                        reply = "继续下一题：请你谈谈你在项目中处理性能瓶颈的一个具体案例。"
                        message_placeholder.markdown(reply)
                    st.session_state.messages.append({"role": "assistant", "content": reply})
                    st.session_state.asked_count += 1
                except Exception as e:  # noqa: BLE001
                    st.error(f"调用大模型失败，请检查 API Key 或网络：{e}")

if not st.session_state.interview_started:
    st.info("请先在左侧完成配置并点击“开始 / 重置面试”。")

```

## 4.2 反思

### 4.2.1 实现的预期功能

 面试配置：目标岗位、难度（简单/普通/地狱）、题数（3-5）。

\- **简历输入：**

\- 支持上传 PDF 简历（优先用于出题）。

\- 支持手动填写简历摘要（兜底）。

\- 上传后在侧边栏显示文件名。

\- PDF 解析失败时自动回退到手动摘要。

\- **面试流程：**

\- 点击“开始/重置面试”后，先由系统询问“准备好了吗？”。

\- 用户回复任意文本或语音后，进入正式技术问答。

\- 每轮一问一答，支持追问。

### 4.2.2 存在的不足与失败情况
